							set_variable = periyar_var
						}

			}

					text = an_agitator_appears_tt

	    				religion = rel:atheist

					        direct
					        brave
					        erudite
					    }
					    on_created = {
							set_variable = periyar_var
						}
					}
				}
			}
		}
	}

				}

					}
				}
			}

# Untouchables and Outcasts
caste_conflict.5 = {
	type = country_event
	placement = root

	duration = 3

	title = caste_conflict.5.t
	desc = caste_conflict.5.d
	flavor = caste_conflict.5.f

	event_image = {
		video = "ip2_intercommunal_tensions"
	}

	dlc = dlc011

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	cooldown = { days = stupidly_long_modifier_time }

	# Triggered by je_dravidian_movement
	trigger = {
		any_scope_character = {
			OR = {
				has_template = BIC_bhimrao_ramji_ambedkar
				has_variable = periyar_var
			}
		}
	}

	immediate = {
		if = {
			limit = {
				any_scope_character = {
					has_template = BIC_bhimrao_ramji_ambedkar
				}
			}
			random_scope_character = {
				limit = {
					has_template = BIC_bhimrao_ramji_ambedkar
				}
				save_scope_as = character_scope
			}
		}
		else = {
			random_scope_character = {
				limit = {
					has_variable = periyar_var
				}
				save_scope_as = character_scope
			}
		}
	}

	cancellation_trigger = {
		NOT = { has_social_hierarchy = british_indian_caste_system }
	}

	option = {
		name = caste_conflict.5.a
		default_option = yes
		scope:character_scope = {
			add_modifier = {
				name = opposition_to_untouchability
				days = normal_modifier_time
			}
		}
		add_radicals = {
			pop_type = clergymen
			value = large_radicals
		}
	}

	option = {
		name = caste_conflict.5.b
		scope:character_scope = {
			add_modifier = {
				name = opposition_to_untouchability
				days = normal_modifier_time
				multiplier = 0.5
			}
		}
		add_modifier = {
			name = priestly_polluted_water
			days = normal_modifier_time
		}
	}
}
