			trigger_event = { id = india_home_rule_events.2 popup = yes }

		name = indian_reform_request_events.1.c

indian_reform_request_events.2 = { # GBR Receives Request for BIC Elected Bureaucrats
	type = country_event
	placement = ROOT

	duration = 3

	title = indian_reform_request_events.2.t
	desc = indian_reform_request_events.2.d
	flavor = indian_reform_request_events.2.f

	event_image = {
		video = "ip2_telegraph_dictate"
	}

	dlc = dlc011

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	# Triggered by indian_reform_request_events.1
	trigger = { }

	cancellation_trigger = {
		c:BIC ?= {
			has_law_or_variant = law_type:law_elected_bureaucrats
		}
	}

	immediate = {
		save_scope_as = overlord_scope
		c:BIC ?= {
			save_scope_as = india_scope
			random_political_lobby = {
				limit = {
					is_political_lobby_type = lobby_indian_national_congress
				}
				save_scope_as = inc_scope
			}
			random_political_movement = {
				limit = {
					is_political_movement_type = movement_india_pan_national
				}
				save_scope_as = indian_national_movement
			}
		}
		set_local_variable = {
			name = india_national_activity
			value = {
				add = scope:indian_national_movement.political_movement_support
				add = scope:indian_national_movement.political_movement_radicalism
			}
		}
	}

	option = {
		name = indian_reform_request_events.2.a
		ai_chance = {
			base = 50
			modifier = {
				trigger = {
					local_var:india_national_activity >= 0.70
				}
				add = -50
			}
			modifier = {
				trigger = {
					any_interest_group = {
						is_in_government = yes
						ig_is_indian_reformist = yes
					}
				}
				add = -25
			}
			modifier = {
				trigger = {

				}
				any_interest_group = {
					is_in_government = yes
					ig_is_indian_imperialist = yes
				}
				add = 25
			}
		}
		c:BIC = {
			indian_reform_denied_effect = yes
			post_notification = overlord_government_reform_denied
		}
		every_interest_group = {
			limit = {
				ig_is_indian_imperialist = yes
				is_marginal = no
			}
			add_modifier = {
    			name = rejected_indian_government_reform_good
    			days = short_modifier_time
    		}
		}
		every_interest_group = {
			limit = {
				ig_is_indian_reformist = yes
				is_marginal = no
			}
			add_modifier = {
    			name = rejected_indian_government_reform_bad
    			days = short_modifier_time
    		}
		}
		if = {
			limit = {
				c:BIC = {
					NOT = {
						has_variable = india_home_rule_failed
					}
				}
				je:je_india_home_rule ?= {
					var:overlord_rejected_reforms < 2
				}
			}
			custom_tooltip = {
				text = indian_national_movement_moves_towards_independence_tt
				je:je_india_home_rule ?= {
					change_variable = {
						name = overlord_rejected_reforms
						add = 1
					}
				}
			}
		}
	}

	option = {
		name = indian_reform_request_events.2.b
		default_option = yes
		ai_chance = {
			base = 50
			modifier = {
				trigger = {
					local_var:india_national_activity >= 0.70
				}
				add = 50
			}
			modifier = {
				trigger = {
					any_interest_group = {
						is_in_government = yes
						ig_is_indian_reformist = yes
					}
				}
				add = 25
			}
			modifier = {
				trigger = {
					any_interest_group = {
						is_in_government = yes
						ig_is_indian_imperialist = yes
					}
				}
				add = -25
			}
		}
		c:BIC = {
			activate_law = law_type:law_elected_bureaucrats
			indian_reform_passed_effect = yes
			post_notification = overlord_government_reform_accepted
		}
		every_interest_group = {
			limit = {
				ig_is_indian_reformist = yes
				is_marginal = no
			}
			add_modifier = {
    			name = approved_indian_government_reform_good
    			days = short_modifier_time
    		}
		}
		every_interest_group = {
			limit = {
				ig_is_indian_imperialist = yes
				is_marginal = no
			}
			add_modifier = {
    			name = approved_indian_government_reform_bad
    			days = short_modifier_time
    		}
		}
		custom_tooltip = indian_nationalism_increase_0_1_per_month_tt
	}
}
