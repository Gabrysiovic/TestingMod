india_home_rule_events.3 = { # Jallianwala Bagh Massacre
	type = country_event
	placement = scope:jallianwala_bagh_state

	duration = 3

	title = india_home_rule_events.3.t
	desc = india_home_rule_events.3.d
	flavor = india_home_rule_events.3.f

	event_image = {
		video = "ep1_redcoats"
	}

	dlc = dlc011

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	cooldown = { days = normal_modifier_time }

	# Triggered by JE

	immediate = {
		je:je_india_home_rule ?= {
			set_variable = jallianwala_massacre
		}
		if = {
			limit = {
				any_scope_state = {
					state_region = s:STATE_HILL_PUNJAB
				}
			}
			random_scope_state = {
				limit = {
					state_region = s:STATE_HILL_PUNJAB
				}
				save_scope_as = jallianwala_bagh_state
				random_scope_culture = {
					limit = {
						has_homeland = prev
					}
					save_scope_as = relevant_culture
				}
			}
		}
		else = {
			random_scope_state = {
				limit = {
					is_in_geographic_region = geographic_region_india
				}
				save_scope_as = jallianwala_bagh_state
				random_scope_culture = {
					limit = {
						has_homeland = prev
					}
					save_scope_as = relevant_culture
				}
			}
		}
		if = {
			limit = {
				game_date > 1900.1.1
				game_date < 1924.1.1
				OR = {
					country_has_primary_culture = cu:british
					overlord ?= {
						country_has_primary_culture = cu:british
					}
				}
			}
			create_character = {
				template = BIC_reginald_dyer
				on_created = {
					set_character_busy_and_immortal = yes
					place_character_in_void = 2400
					set_variable = dyer_variable
				}
			}
		}
		else = {
			create_character = {
				is_general = yes
				interest_group = ig_armed_forces
				ideology = ideology_authoritarian
				traits = {
					war_criminal
					cruel
					plains_commander
				}
				on_created = {
					set_character_busy_and_immortal = yes
					place_character_in_void = 2400
					set_variable = dyer_variable
				}
			}
		}
		random_character_in_void = {
			limit = {
				has_variable = dyer_variable
			}
			save_scope_as = dyer_scope
		}
		random_political_movement = {
			limit = {
				is_political_movement_type = movement_india_pan_national
			}
			save_scope_as = indian_national_movement
		}
	}

	option = {
		name = india_home_rule_events.3.a
		default_option = yes
		scope:jallianwala_bagh_state ?= {
			kill_population_in_state = {
				value = 500
				culture = scope:relevant_culture
			}
			add_radicals_in_state = {
				value = large_radicals
				culture = scope:relevant_culture
			}
		}
		add_modifier = {
			name = modifier_india_massacred_protesters
			days = long_modifier_time
			is_decaying = yes
		}
		random_political_lobby = {
			limit = {
				is_political_lobby_type = lobby_indian_national_congress
			}
			change_appeasement = {
				amount = -3
				factor = appeasement_indian_hostile_action
			}
		}
		trigger_event = { id = india_home_rule_events.4 days = 2 }
		if = {
			limit = {
				is_subject = yes
				NOT = {
					has_variable = india_home_rule_failed
				}
			}
			custom_tooltip = indian_national_movement_moves_towards_independence_tt
			overlord ?= {
				post_notification = jallianwala_massacre
			}
		}
	}
}
