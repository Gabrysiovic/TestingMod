# These events are now just used to trigger je_british_dictates; if incorporated into the base game they can be
# consolidated into a single event. There should curerntly be a 25% chance each year of trigger one of these
# events (and thus the JE). They will only trigger *after* Unstable Raj / Sepoy Mutiny.
british_dictates.1 = {
	type = country_event

	hidden = yes

	trigger = {
		has_dlc_feature = ip2_content
		c:BIC ?= this
		NOR = {
			has_journal_entry = je_uneasy_raj
			has_journal_entry = je_sepoy_mutiny
		}
		NOT = {
			has_variable = british_dictates_started
		}
	}

	immediate = {
		set_variable = british_dictates_started
	}
}

# New Events for Reworked Colonial Dictates JEs

british_dictates.5 = { # Opening Event for Colonial Dictates
	type = country_event
	placement = root

	title = british_dictates.5.t
	desc = british_dictates.5.d
	flavor = british_dictates.5.f

	event_image = {
		video = "europenorthamerica_capitalists_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	duration = 3

	cooldown = { days = never_fire_again_modifier_time }

	# Triggered by je_british_dictates

	cancellation_trigger = {
		NOT = {
			c:BIC ?= ROOT
		}
	}

	immediate = {
		if = {
			limit = {
				is_subject = yes
			}
			overlord ?= {
				save_scope_as = overlord_scope
			}
		}
		else = {
			save_scope_as = overlord_scope
		}
	}

	option = {
		name = british_dictates.5.a
		default_option = yes
		custom_tooltip = je_british_dictates_activated_tt
	}
}

british_dictates.6 = { # Colonial Dictates Success
	type = country_event
	placement = root

	title = british_dictates.6.t
	desc = british_dictates.6.d
	flavor = british_dictates.6.f

	event_image = {
		video = "unspecific_ruler_speaking_to_people"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	duration = 3

	cooldown = { days = never_fire_again_modifier_time }

	# Triggered by je_british_dictates

	cancellation_trigger = {
		NOT = {
			c:BIC ?= ROOT
		}
	}

	option = {
		name = british_dictates.6.a
		default_option = yes
		if = {
			limit = {
				is_subject = yes
			}
			change_relations = {
				country = overlord
				value = 30
			}
		}
		add_modifier = {
			name = india_model_colony
			days = long_modifier_time
		}
		if = {
			limit = {
				# Keep rewards from showing in je_british_dictates until sub-jounral completed
				NOT = {
					has_journal_entry = je_british_dictate_plantations
				}
			}
			if = {
				limit = {
					has_variable = plantations_prioritize_construction
				}
				add_modifier = colonial_prosperity_plantation_construction
			}
			else_if = {
				limit = {
					has_variable = plantations_prioritize_productivity
				}
				add_modifier = colonial_prosperity_plantation_productivity
			}
		}
		if = {
			limit = {
				# Keep rewards from showing in je_british_dictates until sub-jounral completed
				NOT = {
					has_journal_entry = je_british_dictate_military
				}
			}
			if = {
				limit = {
					has_variable = military_prioritize_army
				}
				add_modifier = colonial_prosperity_army
			}
			else_if = {
				limit = {
					has_variable = military_prioritize_navy
				}
				add_modifier = colonial_prosperity_navy
			}
		}
		if = {
			limit = {
				# Keep rewards from showing in je_british_dictates until sub-jounral completed
				NOT = {
					has_journal_entry = je_british_dictate_universities
				}
			}
			if = {
				limit = {
					has_variable = universities_prioritize_higher_education
				}
				add_modifier = colonial_prosperity_higher_education
			}
			else_if = {
				limit = {
					has_variable = universities_prioritize_public_schools
				}
				add_modifier = colonial_prosperity_public_schools
			}
		}
	}
}

british_dictates.7 = { # Timeout Event for Colonial Dictates
	type = country_event
	placement = root

	title = british_dictates.7.t
	desc = british_dictates.7.d
	flavor = british_dictates.7.f

	event_image = {
		video = "ep1_clandestine_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	duration = 3

	cooldown = { days = never_fire_again_modifier_time }

	# Triggered by je_british_dictates

	cancellation_trigger = {
		NOT = {
			c:BIC ?= ROOT
		}
	}

	immediate = {
		if = {
			limit = {
				is_subject = yes
			}
			overlord ?= {
				save_scope_as = overlord_scope
			}
		}
		else = {
			save_scope_as = overlord_scope
		}
	}

	option = {
		name = british_dictates.7.a
		default_option = yes
		if = {
			limit = {
				NOR = {
					has_variable = british_dictate_plantations_complete
					has_variable = british_dictate_military_complete
					has_variable = british_dictate_universities_complete
				}
			}
			if = {
				limit = {
					is_subject = yes
				}
				change_relations = {
					country = overlord
					value = -30
				}
			}
			add_modifier = {
				name = india_incomplete_dictates
				days = long_modifier_time
				multiplier = 2
			}
		}
		else_if = {
			limit = {
				calc_true_if = {
					amount <= 1
					has_variable = british_dictate_plantations_complete
					has_variable = british_dictate_military_complete
					has_variable = british_dictate_universities_complete
				}
			}
			if = {
				limit = {
					is_subject = yes
				}
				change_relations = {
					country = overlord
					value = -15
				}
			}
			add_modifier = {
				name = india_incomplete_dictates
				days = long_modifier_time
				multiplier = 1
			}
		}
	}
}
