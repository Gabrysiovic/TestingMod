					add_modifier = {
						name = modifier_jailed_movement_agitator
						days = normal_modifier_time
						multiplier = jail_three_year_value
					}
				}
			}
		}
		else = {
			owner = {
				add_radicals = {
					value = 0.01

				}
			}
		}
		scope:prisoner_scope = {
			custom_tooltip = {
				text = to_jail_3_tt
				set_variable = { name = prison_time_var value = 36 }
