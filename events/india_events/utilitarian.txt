# Good Governance For All, on JE completion
utilitarian.1 = {
	type = country_event
	placement = root
	title = utilitarian.1.t
	desc = utilitarian.1.d
	flavor = {
		first_valid = {
			triggered_desc = {
				desc = utilitarian.1.f # Loc if a Utilitarian character exists
				trigger = {
					any_scope_character = {
						has_ideology = ideology:ideology_utilitarian_leader
					}
				}
			}
			triggered_desc = {
				desc = utilitarian.1.f2 # Loc if no Utilitarian character exists
				trigger = {
					always = yes
				}
			}
		}
	}

	event_image = {
		video = "ip2_india_factory_floor"
	}

	dlc = dlc011

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	duration = 3

	cooldown = { days = stupidly_long_modifier_time }

	# Triggered by completion of je_imperialism_of_promise
	trigger = {}

	immediate = {
		if = {
			limit = {
				any_scope_character = {
					has_template = agitator_john_stuart_mill
				}
			}
			random_scope_character ?= {
				limit = {
					has_template = agitator_john_stuart_mill
				}
				save_scope_as = relevant_character
			}
		}
		else = {
			random_scope_character = {
				limit = {
					has_ideology = ideology:ideology_utilitarian_leader
				}
				save_scope_as = relevant_character
			}
		}
		random_interest_group = {
			limit = {
				is_interest_group_type = ig_industrialists
			}
			save_scope_as = relevant_ig
		}
		random_scope_character = {
			limit = {
				has_template = agitator_john_stuart_mill
			}
			# His work complete, John Stuart Mill walks into the sunset.
			set_career_length = {
				years = 0.25
				random_range = { 1.0 4.0 } # 0.25-1 years
			}
		}
	}

	cancellation_trigger = { }

	option = {
		name = utilitarian.1.a
		default_option = yes
		ig:ig_industrialists ?= {
			if = {
				limit = {
					has_ideology = ideology:ideology_malthusian
				}
				remove_ideology = ideology_malthusian
				add_ideology = ideology_utilitarian_millian
			}
		}
		add_modifier = {
			name = utilitarian_bonus
			days = long_modifier_time
			multiplier = 5
		}
		add_modifier = {
			name = utilitarian_bureaucracy
			multiplier = 5
		}
		every_interest_group = {
			limit = {
				any_scope_character = {
					OR = {
						has_role_of_type = agitator
						has_role_of_type = politician
					}
					has_ideology = ideology:ideology_utilitarian_leader
				}
			}
			add_modifier = {
				name = util_colonial_admin_success
				days = long_modifier_time
				multiplier = 5
			}
		}
	}
}

			if = {
				limit = {
					has_modifier = persuasive_util_pamphlet_pol_str

			}
			add_modifier = {
				name = util_colonial_admin_fail
				days = long_modifier_time
			}
		}
		random_political_movement = {
			limit = {
				is_political_movement_type = movement_utilitarian
				has_modifier = persuasive_util_pamphlet_attract
			}
			remove_modifier = persuasive_util_pamphlet_attract
		}
		if = {
			limit = {
				exists = scope:john_stuart_mill_scope
			}
			custom_tooltip = {
				text = john_stuart_mill_busy_tt
				scope:john_stuart_mill_scope ?= {
					if = {
						limit = {
							has_modifier = utilitarian_advocate
						}
						remove_modifier = utilitarian_advocate
					}
					if = {
						limit = {
							has_modifier = champion_of_utilitarianism
						}
						remove_modifier = champion_of_utilitarianism
					}
					place_character_in_void = 360
				}
			}
		}
		every_scope_character = {
			limit = {
				has_ideology = ideology:ideology_utilitarian_leader
				NOT = {
					has_template = agitator_john_stuart_mill
				}
			}
			retire_character = yes
		}
	}
}

				add = 1
			}

		}
