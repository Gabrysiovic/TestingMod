india_protest.2 = { # Farmers on Strike
	type = country_event
	placement = scope:farmer_strike_state

	duration = 3

	title = india_protest.2.t
	desc = india_protest.2.d
	flavor = india_protest.2.f

	event_image = {
		video = "gfx/event_pictures/ip2_india_countryside.bk2"
	}

	dlc = dlc011

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	# Generic peaceful protest event for je_india_home_rule
	trigger = {
		any_scope_state = {
			NOT = { has_modifier = indian_farmers_on_strike }
			is_in_geographic_region = geographic_region_india
			state_has_building_group_levels = {
 				type = bg_plantations
 				value >= 10
			}
		}
	}

	immediate = {
		ordered_scope_state = {
			limit = {
				NOT = { has_modifier = indian_farmers_on_strike }
				is_in_geographic_region = geographic_region_india
				state_has_building_group_levels = {
 					type = bg_plantations
 					value >= 10
				}
			}
			position = 0
			order_by = {
				value = 0
				every_scope_building = {
					limit = {
						is_building_group = bg_plantations
					}
					prev = {
						add = prev.level
					}
				}
			}
			save_scope_as = farmer_strike_state
		}
		random_political_movement = {
			limit = {
				is_political_movement_type = movement_india_pan_national
			}
			save_scope_as = indian_independence_movement
		}
		if = {
			limit = {
				is_subject = yes
			}
			overlord ?= {
				save_scope_as = overlord_scope
			}
		}
		else = {
			save_scope_as = overlord_scope
		}
		g:dye = {
			save_scope_as = dye_scope
		}
		g:fabric = {
			save_scope_as = fabric_scope
		}
		g:opium = {
			save_scope_as = opium_scope
		}
	}

	option = {
		name = india_protest.2.a
		default_option = yes
		ai_chance = 5
		ordered_scope_state = {
			limit = {
				NOT = { has_modifier = indian_farmers_on_strike }
				is_in_geographic_region = geographic_region_india
				state_has_building_group_levels = {
 					type = bg_plantations
 					value >= 10
				}
			}
			min = 0
			max = 3
			check_range_bounds = no
			order_by = {
				value = 0
				every_scope_building = {
					limit = {
						is_building_group = bg_plantations
					}
					prev = {
						add = prev.level
					}
				}
			}
			add_modifier = {
				name = indian_farmers_on_strike
				days = normal_modifier_time
				is_decaying = yes
			}
		}
		custom_tooltip = {
			text = indian_nationalism_advance_5_tt
			je:je_india_home_rule ?= {
				add_progress = { value = 5 name = je_india_home_rule_progressbar }
			}
		}
		if = {
			limit = {
				has_journal_entry = je_india_non_cooperation_movement
			}
			custom_tooltip = {
				text = increase_peaceful_protest_action_tt
				change_variable = {
					name = indian_peaceful_protest_actions
					add = 1
				}
			}
		}
	}

	option = {
		name = india_protest.2.b
		ai_chance = 0
		add_small_indian_radicals = yes
	}
}
