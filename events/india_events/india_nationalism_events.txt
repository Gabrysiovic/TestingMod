india_nationalism_events.4 = { # Gandhi in Sooth Efrika
	type = country_event
	placement = scope:gandhi_state

	duration = 3

	title = india_nationalism_events.4.t
	desc = india_nationalism_events.4.d
	flavor = india_nationalism_events.4.f

	gui_window = event_window_1char_tabloid

	left_icon = scope:gandhi_scope

	dlc = dlc011

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	cooldown = { days = normal_modifier_time }

	# Triggered by on_yearly_events
	trigger = {
		has_dlc_feature = ip2_content
		game_date >= 1893.1.1
		# So Gandhi doesn't spend an insignificant amount of time in South Africa
		game_date <= 1895.1.1
		ROOT = {
			NOT = {
				c:BIC ?= this
			}
			is_subject_of = c:BIC.overlord
			is_country_type = colonial
		}
		NOR = {
			is_in_geographic_region = geographic_region_india
			is_in_geographic_region = geographic_region_europe
			is_in_geographic_region = geographic_region_north_america
		}
		any_scope_state = {
			any_scope_pop = {
				culture = {
					has_discrimination_trait_group = heritage_group_south_asian
				}
			}
		}
		NOT = {
			has_law_or_variant = law_type:law_multicultural
		}
		NOT = {
			has_global_variable = gandhi_spawn
		}
	}

	immediate = {
		set_global_variable = gandhi_spawn
		if = {
			limit = {
				any_scope_state = {
					state_region = s:STATE_ZULULAND
					any_scope_pop = {
						culture = {
							has_discrimination_trait_group = heritage_group_south_asian
						}
					}
				}
			}
			s:STATE_ZULULAND.region_state:SAF ?= {
				save_scope_as = gandhi_state
			}
		}
		else_if = {
			limit = {
				any_scope_state = {
					state_region = s:STATE_EASTERN_CAPE
					any_scope_pop = {
						culture = {
							has_discrimination_trait_group = heritage_group_south_asian
						}
					}
				}
			}
			s:STATE_EASTERN_CAPE.region_state:SAF ?= {
				save_scope_as = gandhi_state
			}
		}
		else = {
			random_scope_state = {
				limit = {
					any_scope_pop = {
						culture = {
							has_discrimination_trait_group = heritage_group_south_asian
						}
					}
				}
				save_scope_as = gandhi_state
			}
		}
		create_character = {
			template = BIC_mohandas_karamchand_gandhi
			on_created = {
				set_variable = gandhi_var
				set_variable = gandhi_south_africa
				# We want Gandhi to be around long enough to be picked up by the Indian Independence Movement
				set_career_length = {
					role = agitator
					years = 25
					random_range = { 1.0 1.166 } # 25-30 years
				}
			}
		}
		random_scope_character = {
			limit = {
				has_variable = gandhi_var
			}
			save_scope_as = gandhi_scope
		}
	}

	option = {
		name = india_nationalism_events.4.a
		default_option = yes
		scope:gandhi_state = {
			add_modifier = {
				name = modifier_natal_indian_congress
				days = long_modifier_time
			}
		}
		every_interest_group = {
			limit = {
				law_enactment_stance = {
					law = law_type:law_multicultural
					value < neutral
				}
				is_marginal = no
			}
			add_modifier = {
				name = listening_to_indian_congress
				years = 5
			}
		}
	}

	option = {
		name = india_nationalism_events.4.b
		scope:gandhi_state = {
			add_modifier = {
				name = modifier_natal_indian_congress
				days = long_modifier_time
				multiplier = 0.5
			}
		}
		scope:gandhi_scope = {
			add_trait = scarred
		}
	}
}

india_nationalism_events.5 = { # Gandhi in India
	type = country_event
	placement = ROOT

	duration = 3

	title = india_nationalism_events.5.t
	desc = india_nationalism_events.5.d
	flavor = india_nationalism_events.5.f

	gui_window = event_window_1char_tabloid

	left_icon = scope:gandhi_scope

	dlc = dlc011

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	cooldown = { days = stupidly_long_modifier_time }

	trigger = {
		has_dlc_feature = ip2_content
		c:BIC ?= this
		has_journal_entry = je_india_home_rule
		OR = {
			AND = {
				# So Gandhi can spend at least some time abroad
				game_date >= 1898.1.1
				je:je_india_home_rule = {
					var:gandhi_arrival_threshold <= "scripted_bar_progress(je_india_home_rule_progressbar)"
				}
			}
			game_date >= 1915.1.1
		}
		NOT = {
			any_country = {
				any_scope_character = {
					has_template = BIC_mohandas_karamchand_gandhi
				}
				is_enacting_law = law_type:law_multicultural
			}
		}
		NOT = {
			has_variable = gandhi_arrived
		}
	}

	immediate = {
		set_variable = gandhi_arrived
		if = {
			limit = {
				any_country = {
					any_scope_character = {
						has_template = BIC_mohandas_karamchand_gandhi
					}
				}
			}
			random_country = {
				limit = {
					any_scope_character = {
						has_template = BIC_mohandas_karamchand_gandhi
					}
				}
				save_scope_as = gandhi_exit_scope
				random_scope_character = {
					limit = {
						has_template = BIC_mohandas_karamchand_gandhi
					}
					transfer_character = ROOT
				}
			}
		}
		else_if = {
			limit = {
				any_character_in_exile_pool = {
					has_template = BIC_mohandas_karamchand_gandhi
				}
			}
			random_character_in_exile_pool = {
				limit = {
					has_template = BIC_mohandas_karamchand_gandhi
				}
				transfer_character = ROOT
			}
		}
		else_if = {
			limit = {
				any_character_in_jail = {
					has_template = BIC_mohandas_karamchand_gandhi
				}
			}
			random_character_in_jail = {
				limit = {
					has_template = BIC_mohandas_karamchand_gandhi
				}
				free_character_from_void = yes
				owner ?= {
					save_scope_as = gandhi_exit_scope
				}
				transfer_character = ROOT
				remove_variable = prison_country
				remove_variable = prison_time_var
			}
		}
		else = {
			create_character = {
				template = BIC_mohandas_karamchand_gandhi
				on_created = {
					set_variable = gandhi_var
				}
			}
		}
		random_scope_character = {
			limit = {
				has_template = BIC_mohandas_karamchand_gandhi
			}
			save_scope_as = gandhi_scope
			# We want Gandhi to be around for the entire Indian Independence Movement
			set_career_length = {
				role = agitator
				years = 40
				random_range = { 1 1.5 } # 40-60 years
			}
			## Drip
			if = {
				limit = {
					has_variable = gandhi_south_africa
				}
				remove_variable = gandhi_south_africa
			}
			# Gandhi will change to his iconic look after 5 years in India
			set_variable = {
				name = gandhi_cap
				years = 5
			}
			##
		}
		if = {
			limit = {
				exists = scope:gandhi_exit_scope
			}
			scope:gandhi_exit_scope = {
				post_notification = gandhi_leaves
			}
		}
		random_political_movement = {
			limit = {
				is_political_movement_type = movement_india_pan_national
			}
			save_scope_as = indian_national_movement
		}
	}

	option = {
		name = india_nationalism_events.5.a
		default_option = yes
		scope:gandhi_scope = {
			add_modifier = {
				name = modifier_father_of_satyagraha
			}
		}
	}
}
