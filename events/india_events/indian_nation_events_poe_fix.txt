indian_nation_events.1 = { # Indian National Congress Founded
	type = country_event
	placement = root

	duration = 3

	title = indian_nation_events.1.t
	desc = indian_nation_events.1.d
	flavor = indian_nation_events.1.f

	event_image = {
		video = "ip2_india_urban_scene"
	}

	dlc = dlc011

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	cooldown = { days = stupidly_long_modifier_time }

	# Triggered by Indian Nation JE

	immediate = {
		je:je_india_home_rule = {
			set_variable = inc_founded
		}
		ig:ig_intelligentsia ?= {
			save_scope_as = intelligentsia_ig
		}
		ig:ig_petty_bourgeoisie ?= {
			save_scope_as = petty_bourgeoisie_ig
		}
		if = {
			limit = {
				NOT = {
					any_political_lobby = {
						is_political_lobby_type = lobby_indian_national_congress
					}
				}
			}
			create_political_lobby = {
				type = lobby_indian_national_congress
				target = ROOT
				add_interest_group = ig:ig_intelligentsia
				add_interest_group = ig:ig_petty_bourgeoisie
				lobby_formation_reason = diplomacy
			}
		}
		else = {
			random_political_lobby = {
				limit = {
					is_political_lobby_type = lobby_indian_national_congress
				}
				if = {
					limit = {
						NOT = {
							any_lobby_member = {
								is_interest_group_type = ig_intelligentsia
							}
						}
					}
					add_lobby_member = ig:ig_intelligentsia
				}
				if = {
					limit = {
						NOT = {
							any_lobby_member = {
								is_interest_group_type = ig_petty_bourgeoisie
							}
						}
					}
					add_lobby_member = ig:ig_petty_bourgeoisie
				}
			}
		}
		random_political_lobby = {
			limit = {
				is_political_lobby_type = lobby_indian_national_congress
			}
			save_scope_as = relevant_lobby
		}
		random_political_movement = {
			limit = {
				is_political_movement_type = movement_india_pan_national
			}
			save_scope_as = indian_national_movement
		}
	}

	option = {
		name = indian_nation_events.1.a
		default_option = yes
		custom_tooltip = independence_lobby_formed_tt
		scope:intelligentsia_ig = {
			add_modifier = {
				name = modifier_indian_national_movement_ig
				days = normal_modifier_time
			}
		}
		scope:petty_bourgeoisie_ig = {
			add_modifier = {
				name = modifier_indian_national_movement_ig
				days = normal_modifier_time
			}
		}
	}

	option = {
		name = indian_nation_events.1.b
		custom_tooltip = independence_lobby_formed_tt
		scope:intelligentsia_ig = {
			add_modifier = {
				name = modifier_indian_national_movement_ig
				days = normal_modifier_time
				multiplier = 2
			}
		}
		scope:petty_bourgeoisie_ig = {
			add_modifier = {
				name = modifier_indian_national_movement_ig
				days = normal_modifier_time
				multiplier = 2
			}
		}
		scope:relevant_lobby = {
			change_appeasement = {
				amount = 1
				factor = appeasement_lobby_gain
			}
		}
		every_interest_group = {
			limit = {
				OR = {
					is_interest_group_type = ig_industrialists
					is_interest_group_type = ig_armed_forces
				}
			}
			add_modifier = {
				name = listening_to_indian_congress
				days = normal_modifier_time
			}
		}
	}
}

indian_nation_events.7 = { # Partition of India
	type = country_event
	placement = ROOT

	duration = 3

	title = indian_nation_events.7.t
	desc = indian_nation_events.7.d
	flavor = indian_nation_events.7.f

	event_image = {
		video = "asia_poor_people_moving"
	}

	dlc = dlc011

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	minor_left_icon = ROOT
	minor_right_icon = scope:pakistan_scope

	cooldown = { days = stupidly_long_modifier_time }

	trigger = {
		has_dlc_feature = ip2_content
		this = {
			OR = {
				has_state_in_state_region = STATE_PUNJAB
				has_state_in_state_region = STATE_SINDH
				has_state_in_state_region = STATE_KASHMIR
				has_state_in_state_region = STATE_PASHTUNISTAN
				has_state_in_state_region = STATE_BALUCHISTAN
				has_state_in_state_region = STATE_QUETTA
				has_state_in_state_region = STATE_EAST_BENGAL
			}
		}
	}

	immediate = {
		save_scope_as = india_scope
		partition_india = yes
		# TODO: Investigate if we just move this into the message
		scope:pakistan_scope ?= {
			set_variable = partitioned_pakistan
		}
		# Make Pakistan a Dominion of Britain if India is also a British subject still
		if = {
			limit = {
				is_subject = yes
			}
			overlord ?= {
				scope:pakistan_scope = {
					set_country_type = colonial
				}
				create_diplomatic_pact = {
					country = scope:pakistan_scope
					type = dominion
				}
			}
		}
		# Partition Migration (India->Pakistan)
		every_scope_state = {
			kill_population_percent_in_state = {
				religion = rel:sunni
				percent = 0.02
			}
			kill_population_percent_in_state = {
				religion = rel:shiite
				percent = 0.02
			}
			kill_population_percent_in_state = {
				religion = rel:ibadi
				percent = 0.02
			}
			every_scope_pop = {
				limit = {
					OR = {
						religion = rel:sunni
						religion = rel:shiite
						religion = rel:ibadi
					}
				}
				random_list = {
					82 = {}
					18 = {
