# The Jewel in the Crown
raj.1 = {
	type = country_event
	placement = ROOT

	title = raj.1.t
	desc = {
		triggered_desc = {
			desc = raj.1.d
			trigger = {
				NOT = {
					has_local_variable = bic_triggered_raj
				}
			}
		}
		triggered_desc = {
			desc = raj.1.d_company
			trigger = {
				has_local_variable = bic_triggered_raj
			}
		}
	}
	flavor = raj.1.f

	duration = 3

	cooldown = { days = never_fire_again_modifier_time }

	event_image = {
		video = "ep1_transfer_of_authority"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	minor_left_icon = c:GBR
	minor_right_icon = c:BIC

	# Triggered by completing je_sepoy_mutiny or by BIC or overlord taking decision to end company rule
	trigger = {
	}

	immediate = {
		c:BIC ?= {
			save_scope_as = company_country
		}
		# TODO: This seems like it's used to affect GBR AI strategy and to control whether the Company->Raj decision is. After we look at those things, see if this is necessary or not.
		# Probably don't need it for Decision but maybe we'll keep for AI strategy
		set_variable = {
			name = british_raj_var
		}
	}

	option = {
		name = raj.1.a
		default_option = yes
		custom_tooltip = activate_british_raj
		# Remove the East India Company's ownership of the BIC country and then disband the company
		c:GBR ?= {
			if = {
				limit = {
					any_company = {
						is_company_type = company_type:company_east_india_company
					}
				}
				hidden_effect = {
					if = {
						limit = {
							c:BIC ?= {
								is_subject_type = subject_type_chartered_company
							}
						}
						random_company = {
							limit = {
								is_company_type = company_type:company_east_india_company
							}
							remove_owned_country = c:BIC
						}
					}
				}
				remove_company = company_type:company_east_india_company
			}
		}
		show_as_tooltip = {
			c:BIC ?= {
				make_bic_raj = yes
			}
		}
		hidden_effect = {
			c:BIC ?= {
				trigger_event = { id = raj.7 popup = yes }
			}
		}
	}
}

raj.2 = {
	type = country_event
	placement = ROOT

	title = raj.2.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = raj.2.d1
				trigger = {
					ruler = {
						age > 60
					}
					NOT = {
						exists = heir
					}
				}
			}
			triggered_desc = {
				desc = raj.2.d2
				trigger = {
					always = yes
				}
			}
		}
	}
	flavor = raj.2.f

	duration = 3

	event_image = {
		video = "ep1_persian_court"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	minor_left_icon = c:BIC

	immediate = {
		c:BIC ?= {
			save_scope_as = overlord_scope
		}
	}

	option = {
		name = raj.2.a
		default_option = yes
		ai_chance = {
			base = 100
			modifier = {
				trigger = {
					liberty_desire > 25
				}
				add = -25
			}
			modifier = {
				trigger = {
					liberty_desire > 75
				}
				add = -75
			}
			modifier = {
				trigger = {
					any_country = {
						any_scope_treaty = {
							binds = ROOT
							any_scope_article = {
								has_type = guarantee_independence
							}
						}
					}
				}
				add = -100
			}
		}
		if = {
			limit = { # Should never happen, but just to be sure.
				is_player = yes
			}
			play_as = scope:overlord_scope
		}
		if = {
			limit = {
				gold_reserves > 0
			}
			custom_tooltip = {
				text = transfers_treasury_to_eic_tt
				scope:overlord_scope ?= {
					add_treasury = {
						value = prev.gold_reserves
					}
				}
			}
		}
		scope:overlord_scope ?= {
			annex = ROOT
		}
	}

	option = {
		name = raj.2.b
		ai_chance = {
			base = 0
			modifier = {
				trigger = {
					liberty_desire > 25
				}
				add = 25
			}
			modifier = {
				trigger = {
					liberty_desire > 75
				}
				add = 75
			}
			modifier = {
				trigger = {
					any_country = {
						any_scope_treaty = {
							binds = ROOT
							any_scope_article = {
								has_type = guarantee_independence
							}
						}
					}
				}
				add = 100
			}
		}
		create_diplomatic_play = {
			target_country = scope:overlord_scope
			type = dp_independence
			add_war_goal = { holder = scope:overlord_scope target_country = ROOT type = annex_country }
		}
	}
}

		NOT = { has_variable = suspected_conversions }

			any_scope_pop = {
				OR = {
					religion = rel:hindu

					religion = rel:shiite
				}
				NOT = {
					religion = { shares_heritage_trait_with_state_religion = prev.owner }
				}
			}
			any_scope_pop = {
				is_pop_type = clergymen
				religion = { shares_heritage_trait_with_state_religion = prev.owner }
			}
		}
	}

	immediate = {
		set_variable = {
			name = suspected_conversions
			years = 12.5
		}
		random_scope_state = {
			limit = {
				any_scope_pop = {
					OR = {
						religion = rel:hindu
						religion = rel:sunni
						religion = rel:shiite
					}
					NOT = {
						religion = { shares_heritage_trait_with_state_religion = prev.owner }
					}
				}
				any_scope_pop = {
					is_pop_type = clergymen
					religion = { shares_heritage_trait_with_state_religion = prev.owner }
				}
			}
			save_scope_as = missionaries_suspicion_state
			random_scope_pop = {
				limit = {
					OR = {
						religion = rel:hindu
						religion = rel:sunni
						religion = rel:shiite
					}
					NOT = {
						religion = { shares_heritage_trait_with_state_religion = prev.owner }
					}
				}
				save_scope_as = not_suspected_missionaries_pop
			}
			random_scope_pop = {
				limit = {
					is_pop_type = clergymen
					religion = { shares_heritage_trait_with_state_religion = prev.owner }
				}
				religion = { save_scope_as = suspected_missionaries }
				save_scope_as = suspected_missionaries_pop
			}
		}
	}

	option = {
		name = raj.5.a
		ai_chance = 1
		scope:missionaries_suspicion_state = {
			# state_conversion_mult
			add_modifier = {
				name = promised_no_proselytization
				days = normal_modifier_time
			}
		}
		if = {
			ig:ig_devout ?= {
				add_modifier = {
					name = promised_no_missionaries
					days = normal_modifier_time
				}
			}
		}
		scope:missionaries_suspicion_state = {
			add_radicals_in_state = {
				religion = scope:suspected_missionaries_pop.religion
				value = medium_radicals
			}
		}

			name = enforcing_conversino_ban
