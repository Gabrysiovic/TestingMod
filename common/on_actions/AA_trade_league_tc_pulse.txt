# AA_trade_league_tc_pulse.txt
# Cohesion is computed inline in AA_power_bloc_identities.txt.
# This file handles game-start effects for the Zollverein.

on_game_started_after_lobby = {
	on_actions = {
		AA_zollverein_game_start
	}
}

AA_zollverein_game_start = {
	effect = {
		c:PRU ?= {
			if = {
				limit = {
					is_power_bloc_leader = yes
					has_dlc_feature = power_bloc_features
				}
				power_bloc = {
					add_cohesion_number = 50
					add_modifier = {
						name = modifier_AA_zollverein
						years = 15
						is_decaying = yes
					}
				}
			}
		}
	}
}

on_law_activated = {
	on_actions = {
		AA_imposition_removal
	}
}

AA_imposition_removal = {
	effect = {
	owner = {
		if = {
				limit = {
					has_law = scope:article.input_law
				}
				remove_modifier = AA_agreeable_to_imposition
		}
	}
	}
}
