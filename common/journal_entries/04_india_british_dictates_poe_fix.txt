REPLACE_OR_CREATE:je_british_dictates = {
	should_be_pinned_by_default_uninvolved_or_context = yes
	icon = "gfx/interface/icons/invention_icons/corporate_charters.dds"

	group = je_group_british_india

	is_shown_when_inactive = {
		has_dlc_feature = ip2_content
		c:BIC ?= this
		NOR = {
			has_journal_entry = je_uneasy_raj
			has_journal_entry = je_sepoy_mutiny
			has_variable = sepoy_mutiny_failed
		}
	}

	possible = {
		custom_tooltip = {
			text = british_dictates_officials_put_forth_plan_tt
			has_variable = british_dictates_started
		}
	}

	timeout = 7300

	immediate = {
		trigger_event = { id = british_dictates.5 popup = yes }
		add_journal_entry = { type = je_british_dictate_plantations }
		add_journal_entry = { type = je_british_dictate_military }
		add_journal_entry = { type = je_british_dictate_universities }
	}

	complete = {
		custom_tooltip = {
			text = je_british_dictate_plantations_complete_tt
			has_variable = british_dictate_plantations_complete
		}
		custom_tooltip = {
			text = je_british_dictate_military_complete_tt
			has_variable = british_dictate_military_complete
		}
		custom_tooltip = {
			text = je_british_dictate_universities_complete_tt
			has_variable = british_dictate_universities_complete
		}
	}

	on_complete = {
		hidden_effect = {
			trigger_event = { id = british_dictates.6 popup = yes }
		}
		show_as_tooltip = {
			execute_event_option = { event = british_dictates.6 option = 0 }
		}
	}

	on_timeout = {
		hidden_effect = {
			trigger_event = { id = british_dictates.7 popup = yes }
		}
		show_as_tooltip = {
			execute_event_option = { event = british_dictates.7 option = 0 }
		}
		if = {
			limit = {
				calc_true_if = {
					amount <= 1
					has_variable = british_dictate_plantations_complete
					has_variable = british_dictate_military_complete
					has_variable = british_dictate_universities_complete
				}
			}
			custom_tooltip = je_british_dictates_consequences_scaled_tt
		}
	}

	invalid = {
		NOT = {
			c:BIC ?= this
		}
	}

	weight = 904
}

REPLACE:je_british_dictate_plantations = {
		
	should_be_pinned_by_default_uninvolved_or_context = yes
	icon = "gfx/interface/icons/invention_icons/intensive_agriculture.dds"

	group = je_group_british_india

	scripted_button = je_british_dictate_plantations_prioritize_construction
	scripted_button = je_british_dictate_plantations_deprioritize_construction
	scripted_button = je_british_dictate_plantations_prioritize_productivity
	scripted_button = je_british_dictate_plantations_deprioritize_productivity

	immediate = {
		set_variable = {
			name = arable_land_percent_goal
			value = 0.30
		}
		update_je_british_dictate_plantations_vars = yes
	}

	on_monthly_pulse = {
		effect = {
			update_je_british_dictate_plantations_vars = yes
		}
	}

	complete = {
		custom_tooltip = {
			text = je_british_dictate_plantations_completion_condition_tt
			var:num_plantations >= var:target_plantations
		}
		trigger_if = {
			limit = {
				has_variable = plantations_prioritize_construction
			}
			calc_true_if = {
				amount >= 2
				leading_producer_of = g:opium
				leading_producer_of = g:tea
				leading_producer_of = g:dye
				leading_producer_of = g:fabric
				leading_producer_of = g:silk
			}
		}
		trigger_else_if = {
			limit = {
				has_variable = plantations_prioritize_productivity
			}
			calc_true_if = {
				amount >= 4
				leading_producer_of = g:opium
				leading_producer_of = g:tea
				leading_producer_of = g:dye
				leading_producer_of = g:fabric
				leading_producer_of = g:silk
			}
		}
		trigger_else = {
			calc_true_if = {
			amount >= 3
				leading_producer_of = g:opium
				leading_producer_of = g:tea
				leading_producer_of = g:dye
				leading_producer_of = g:fabric
				leading_producer_of = g:silk
			}
		}
		custom_tooltip = {
			text = colonial_dictates_timer_tt
			scope:journal_entry ?= {
				journal_entry_age > 730
			}
		}
	}

	on_complete = {
		set_variable = british_dictate_plantations_complete
		hidden_effect = {
			trigger_event = { id = british_dictates.8 }
		}
		show_as_tooltip = {
			execute_event_option = { event = british_dictates.8 option = 0 }
		}
	}

	fail = {
		custom_tooltip = {
			text = british_dictates_time_run_out_tt
			NOT = {
				has_journal_entry = je_british_dictates
			}
		}
	}

	invalid = {
		NOT = {
			c:BIC ?= this
		}
	}

	weight = 903
}
