REPLACE:je_india_non_cooperation_movement = {
	should_be_pinned_by_default_uninvolved_or_context = yes
	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	group = je_group_british_india

	is_shown_when_inactive = {
		has_dlc_feature = ip2_content
		c:BIC ?= this
		has_journal_entry = je_india_home_rule
	}

	possible = {
		has_journal_entry = je_india_home_rule
		custom_tooltip = {
			text = indian_nationalism_above_175_tt
			je:je_india_home_rule = {
				"scripted_bar_progress(je_india_home_rule_progressbar)" >= 175
			}
		}
	}

	immediate = {
		random_political_movement = {
			limit = {
				is_political_movement_type = movement_india_pan_national
			}
			save_scope_as = indian_national_movement
		}
		set_variable = {
			name = indian_peaceful_protest_actions
			value = 0
		}
		set_variable = {
			name = indian_violent_protest_actions
			value = 0
		}
		trigger_event = { id = india_non_cooperation.1 popup = yes }
		if = {
			limit = {
				is_subject = yes
			}
			overlord ?= {
				post_notification = non_cooperation_movement_started
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = non_cooperation_peaceful_protest_actions_tt
			var:indian_peaceful_protest_actions >= 3
		}
	}

	on_complete = {
		hidden_effect = {
			trigger_event = { id = india_non_cooperation.7 popup = yes }
		}
		show_as_tooltip = {
			execute_event_option = { event = india_non_cooperation.7 option = 0 }
		}
		overlord = {
			post_notification = non_cooperation_movement_successful
		}
	}

	#event_outcome_completed_desc = event_effects_india_non_cooperation.7_tt

	fail = {
		custom_tooltip = {
			text = non_cooperation_violent_protest_actions_tt
			var:indian_violent_protest_actions >= 2
		}
	}

	on_fail = {
		hidden_effect = {
			trigger_event = { id = india_non_cooperation.5 popup = yes }
		}
		show_as_tooltip = {
			execute_event_option = { event = india_non_cooperation.5 option = 0 }
		}
		if = {
			limit = {
				is_subject = yes
			}
			overlord = {
				post_notification = non_cooperation_movement_failed
			}
		}
	}

	invalid = {
		NOT = {
			has_journal_entry = je_india_home_rule
		}
	}
}
