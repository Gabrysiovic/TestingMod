	on_complete = {
		c:BIC ?= {
			if = {
				limit = {
					is_subject = yes
				}
				overlord = {
					# Used for scripted effects
					save_scope_as = overlord_scope
				}
			}
			# Free from jail all Indian National Movement agitators
			if = {
				limit = {
					any_character_in_jail = {
						has_variable = indian_national_movement_prisoner
					}
				}
				custom_tooltip = {
					text = all_indian_national_movement_agitators_freed_tt
					every_character_in_jail = {
						limit = {
							has_variable = indian_national_movement_prisoner
						}
						free_character_from_void = yes
						remove_variable = prison_country
						remove_variable = prison_time_var
					}
				}
			}
			# Make India
			change_tag = BHT
			# Take Indian states and any princely states owned by overlord
			if = {
				limit = {
					exists = scope:overlord_scope
				}
				custom_tooltip = {
					text = all_indian_states_transferred_from_overlord_tt
					transfer_all_indian_states_from_overlord = yes
					transfer_all_princely_state_subjects_from_overlord = yes
				}
			}
			# Change Cultures
			custom_tooltip = {
				text = south_asian_primary_cultures_tt
				grant_indian_regional_cultures = yes
			}
			if = {
				limit = {
					has_variable = india_home_rule_failed
					is_subject = yes
				}
				remove_non_indian_cultures = yes
			}
			# Update Religion
			if = {
				limit = {
					has_variable = indian_partition_promised
					 # Has the necessary land to create Pakistan
					OR = {
						has_state_in_state_region = STATE_PUNJAB
						has_state_in_state_region = STATE_SINDH
						has_state_in_state_region = STATE_KASHMIR
						has_state_in_state_region = STATE_PASHTUNISTAN
						has_state_in_state_region = STATE_BALUCHISTAN
						has_state_in_state_region = STATE_QUETTA
						has_state_in_state_region = STATE_EAST_BENGAL
					}
				}
				set_state_religion = rel:hindu
			} else = {
				change_indian_state_religion = yes
			}
			# Other Effects
			india_independence_ig_effects = yes
			remove_bic_modifiers = yes
			remove_indian_peaceful_protest_effects = yes
			# Get rid of the East India Company in the off-chance it still exists
			scope:overlord_scope ?= {
				if = {
					limit = {
						any_company = {
							is_company_type = company_type:company_east_india_company
						}
					}
					remove_company = company_type:company_east_india_company
				}
			}
			# Change country type if applicable
			if = {
				limit = {
					OR = {
						is_subject = no
						has_variable = india_home_rule_failed
					}
				}
				hidden_effect = {
					set_country_type = recognized
				}
			}
			# Make independent if seeking independence
			if = {
				limit = {
					has_variable = india_home_rule_failed
					is_subject = yes
				}
				make_independent = yes
			}
			# Don't go from Monarchy/Theocracy -> Monarchy/Theocracy
			if = {
				limit = {
					OR = {
						# Includes Colonial Administration
						has_law_or_variant = law_type:law_monarchy
						has_law_or_variant = law_type:law_social_monarchy
						has_law_or_variant = law_type:law_theocracy
					}
				}
				activate_law = law_type:law_parliamentary_republic
			}
			# Remove some/all non-Indian characters
			hidden_effect = {
				if = {
					limit = {
						has_variable = india_home_rule_failed
						is_subject = yes
					}
					remove_non_indian_characters = yes
				}
				else = {
					remove_non_indian_politicians = yes
				}
			}
			hidden_effect = {
				every_political_lobby = {
					limit = {
						OR = {
							is_political_lobby_type = lobby_pro_overlord
							is_political_lobby_type = lobby_indian_national_congress
						}
					}
					disband_political_lobby = yes
				}
			}
			# Recalc IG support and call election
			hidden_effect = {
				recalculate_pop_ig_support = yes
				if = {
					limit = {
						country_has_voting_franchise = yes
					}
					call_election = {
						months = 6
					}
				}
			}
			# Diplomatic Relations and Truce w/ Former Overlord
			if = {
				limit = {
					exists = scope:overlord_scope
				}
				hidden_effect = {
					scope:overlord_scope = {
						set_local_variable = {
							name = relations_set
							value = 0
						}
						if = {
							limit = {
								any_interest_group = {
									is_in_government = yes
									ig_is_indian_imperialist = yes
								}
							}
							change_local_variable = {
								name = relations_set
								subtract = 30
							}
						}
						if = {
							limit = {
								any_interest_group = {
									is_in_government = yes
									ig_is_indian_reformist = yes
								}
							}
							change_local_variable = {
								name = relations_set
								add = 30
							}
						}
						set_relations = {
							country = scope:india_scope
							value = local_var:relations_set
						}
						create_bidirectional_truce = {
							country = scope:india_scope
							months = 60
						}
					}
				}
			}
			# Give Union of India JE
			if = {
				limit = {
					any_direct_subject = {
						is_princely_state = yes
					}
				}
				hidden_effect = {
					add_journal_entry = {
						type = je_federation_of_india
					}
				}
			}
			# Burmese Sovereignty Effects
			indian_sovereignty_je_burma_effects = yes
			# Partition India if Communal Divides failed
			if = {
				limit = {
					has_variable = indian_partition_promised
					# Has the necessary land to create Pakistan
					OR = {
						has_state_in_state_region = STATE_PUNJAB
						has_state_in_state_region = STATE_SINDH
						has_state_in_state_region = STATE_KASHMIR
						has_state_in_state_region = STATE_PASHTUNISTAN
						has_state_in_state_region = STATE_BALUCHISTAN
						has_state_in_state_region = STATE_QUETTA
						has_state_in_state_region = STATE_EAST_BENGAL
					}
				}
				custom_tooltip = {
					text = india_will_partition_tt
					trigger_event = { id = indian_nation_events.7 }
				}
			}
			trigger_event = { id = india_home_rule_events.1 popup = yes }
		}
	}

	on_monthly_pulse = {
		effect = {
			scope:indian_national_movement ?= {
				# Used for calculating AI willingness to request reforms
				scope:journal_entry ?= {
					set_variable = {
						name = india_national_activity
						value = {
							add = prev.political_movement_support
							add = prev.political_movement_radicalism
						}
					}
				}
			}
			c:BIC ?= {
				# Check if India has changed overlord, take appropriate action
				if = {
					limit = {
						is_subject = yes
						overlord = {
							NOT = {
								is_involved_in_journal_entry = je_india_home_rule
							}
						}
					}
					# Remove previous overlord
					every_country = {
						limit = {
							is_involved_in_journal_entry = je_india_home_rule
							NOT = {
								c:BIC ?= this
							}
						}
						scope:journal_entry = {
							remove_involved_country = prev
						}
					}
					# Add new overlord
					scope:journal_entry = {
						add_involved_country = prev.overlord
					}
					# Notification Events
					trigger_event = { id = indian_nation_events.14 }
					overlord = {
						trigger_event = { id = indian_nation_events.13 }
					}
				}
				# Track progress blocked by Imperialist overlord government
				if = {
					limit = {
						is_subject = yes
						NOT = {
							has_variable = india_home_rule_failed
						}
						scope:journal_entry = {
							var:imperialist_blocked_progress < 30
						}
						overlord ?= {
							any_interest_group = {
								is_in_government = yes
								ig_is_indian_imperialist = yes
							}
							NOT = {
								any_interest_group = {
									is_in_government = yes
									ig_is_indian_reformist = yes
								}
							}
						}
					}
					scope:journal_entry = {
						change_variable = {
							name = imperialist_blocked_progress
							add = {
								value = "scripted_bar_progress(je_india_home_rule_progressbar)"
								multiply = 0.001
							}
						}
					}
				}
				# Indian National Congress Founded
				if = {
					limit = {
						scope:journal_entry = {
							NOT = {
								has_variable = inc_founded
							}
							"scripted_bar_progress(je_india_home_rule_progressbar)" >= 10
						}
					}
					trigger_event = { id = indian_nation_events.1 popup = yes }
				}
				# Founding of All-India Muslim League and Hindu Mahasabha
				if = {
					limit = {
						NOT = {
							has_variable = hindu_muslim_nationalist_movements_founded
						}
						je:je_india_home_rule = {
							"scripted_bar_progress(je_india_home_rule_progressbar)" >= 125
						}
					}
					set_variable = hindu_muslim_nationalist_movements_founded
					trigger_event = { id = indian_nation_events.8 popup = no }
				}
				# Rowlatt Act Passed
				if = {
					limit = {
						scope:journal_entry = {
							NOT = {
								has_variable = rowlatt_act
							}
							"scripted_bar_progress(je_india_home_rule_progressbar)" >= 150
						}
						scope:indian_national_movement ?= {
							political_movement_radicalism >= 0.50
						}
					}
					trigger_event = { id = indian_nation_events.2 popup = yes }
				}
				# Jallianwala Massacre
				if = {
					limit = {
						je:je_india_home_rule = {
							has_variable = rowlatt_act
							NOT = {
								has_variable = jallianwala_massacre
							}
						}
						scope:indian_national_movement ?= {
							political_movement_radicalism >= 0.65
						}
					}
					trigger_event = { id = india_home_rule_events.3 popup = yes }
				}
				# Quit India Movement
				if = {
					limit = {
						is_subject = yes
						has_variable = india_home_rule_failed
						scope:journal_entry ?= {
							NOT = {
								has_variable = quit_india_movement_started
							}
							"scripted_bar_progress(je_india_home_rule_progressbar)" >= 450
							"scripted_bar_progress(je_india_home_rule_progressbar)" <= 485
						}
					}
					overlord ?= {
						trigger_event = { id = indian_nation_events.4 popup = yes }
					}
				}
				# INC Declares that it's seeking full independence
				if = {
					limit = {
						is_subject = yes
						NOT = {
							has_variable = india_home_rule_failed
						}
						calc_true_if = {
							amount >= 3
							scope:journal_entry ?= {
								var:overlord_rejected_reforms >= 2
							}
							scope:journal_entry ?= {
								var:imperialist_blocked_progress >= 30
							}
							scope:journal_entry ?= {
								has_variable = rowlatt_act
							}
							scope:journal_entry ?= {
								has_variable = jallianwala_massacre
							}
							scope:journal_entry ?= {
								has_variable = simon_commission_rejects_sovereignty
							}
							scope:journal_entry ?= {
								has_variable = overlord_denied_home_rule
							}
						}
					}
					trigger_event = { id = indian_nation_events.6 months = 3 popup = yes }
					if = {
						limit = {
							is_subject = yes
						}
						overlord ?= {
							trigger_event = { id = indian_nation_events.6 months = 3 popup = yes }
						}
					}
				}
				# Simon Commission Countdown to earliest dispatch date
				if = {
					limit = {
						scope:journal_entry = {
							has_variable = simon_commission_timer_early_var
						}
					}
					scope:journal_entry = {
						change_variable = { name = simon_commission_timer_early_var subtract = 1 }
						if = {
							limit = {
								var:simon_commission_timer_early_var <= 0
							}
							remove_variable = simon_commission_timer_early_var
						}
					}
				}
				# Simon Commission Countdown to forced commission
				if = {
					limit = {
						scope:journal_entry = {
							has_variable = simon_commission_timer_var
						}
					}
					scope:journal_entry = {
						change_variable = { name = simon_commission_timer_var subtract = 1 }
					}
					if = {
						limit = {
							scope:journal_entry = {
								var:simon_commission_timer_var <= 0
							}
						}
						scope:journal_entry = {
							remove_variable = simon_commission_timer_var
						}
						if = {
							limit = {
								is_subject = yes
							}
							overlord ?= {
								trigger_event = { id = india_home_rule_events.10 popup = yes }
							}
						}
					}
				}
				# Burmese Freedom Fighters
				if = {
					limit = {
						NOR = {
							has_variable = opposing_burmese_independence
							has_variable = supporting_burmese_subject
							has_variable = supporting_burmese_independence
						}
						scope:journal_entry = {
							has_variable = inc_founded
						}
					}
					# (If all fervor at 100, triggers at 100 Indian nationalism; at all 50 triggers at 300, etc.)
					set_local_variable = {
						name = burmese_fervor_threshold
						value = {
							value = 500
							subtract = cu:burmese.culture_current_fervor
							subtract = cu:kachin.culture_current_fervor
							subtract = cu:karen.culture_current_fervor
							subtract = cu:mon.culture_current_fervor
						}
					}
					if = {
						limit = {
							scope:journal_entry = {
								"scripted_bar_progress(je_india_home_rule_progressbar)" >= local_var:burmese_fervor_threshold
							}
						}
						trigger_event = {
							id = indian_nation_events.10
						}
					}
				}
				# "Reminder" to petition/demand Indian Home Rule / Independence
				if = {
					limit = {
						scope:journal_entry = {
