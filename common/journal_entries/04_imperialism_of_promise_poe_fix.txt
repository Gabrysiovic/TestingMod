REPLACE:je_imperialism_of_promise = {
	icon = "gfx/interface/icons/invention_icons/central_planning.dds"

	group = je_group_british_india

	is_shown_when_inactive = {
		has_dlc_feature = ip2_content
		c:BIC ?= this
	}

	possible = {
		OR = {
			custom_tooltip = {
				text = utilitarian_character_trigger
				OR = {
					any_scope_character = {
						has_role_of_type = agitator
						has_ideology = ideology:ideology_utilitarian_leader
					}
					any_interest_group = {
						leader ?= {
							has_ideology = ideology:ideology_utilitarian_leader
						}
					}
				}
			}
			any_political_movement = {
				is_political_movement_type = movement_utilitarian
				political_movement_support >= 0.03
			}
		}
	}

	immediate = {
		trigger_event = { id = utilitarian.9 popup = yes }
		ig:ig_industrialists ?= {
			save_scope_as = industrialists_ig_scope
		}
	}

	complete = {
		custom_tooltip = {
			text = bureaucrats_no_shortage_trigger
			bureaucracy >= 0
			approaching_bureaucracy_shortage = no
		}
		has_law_or_variant = law_type:law_public_schools
		institution_investment_level = {
			institution = institution_schools
			value >= 3
		}
		NOT = {
			has_law_or_variant = law_type:law_no_womens_rights
		}
		has_law_or_variant = law_type:law_slavery_banned
		OR = {
			has_law_or_variant = law_type:law_proportional_taxation
			has_law_or_variant = law_type:law_graduated_taxation
		}
		OR = {
			NOR = {
				has_law_or_variant = law_type:law_hindu_caste_codified
				has_law_or_variant = law_type:law_hindu_caste_enforced
			}
			AND = {
				custom_tooltip = {
					text = raj_forced_caste_system_codified_tt
					scope:journal_entry ?= {
						has_variable = caste_system_codified_compelled
					}
					has_law_or_variant = law_type:law_hindu_caste_codified
				}
				NOT = {
					has_law_or_variant = law_type:law_hindu_caste_enforced
				}
			}
		}
	}

	fail = {
		OR = {
			has_journal_entry = je_sepoy_mutiny
			has_civil_war_from_movement_type = movement_type:movement_cultural_minority
			has_civil_war_from_movement_type = movement_type:movement_religious_minority
		}
	}

	invalid = {
		custom_tooltip = {
            text = rejected_imperialism_of_promise_tt
            has_variable = rejected_imperialism_of_promise
        }
	}

	on_complete = {
		hidden_effect = {
			trigger_event = { id = utilitarian.1 popup = yes }
		}
		show_as_tooltip = {
			execute_event_option = { event = utilitarian.1 option = 0 }
		}
		set_variable = utilitarianism_done
	}

	on_fail = {
		hidden_effect = {
			trigger_event = { id = utilitarian.2 }
		}
		show_as_tooltip = {
			execute_event_option = { event = utilitarian.2 option = 0 }
		}
		set_variable = utilitarianism_done
	}

	on_timeout = {
		hidden_effect = {
			trigger_event = { id = utilitarian.3 popup = yes }
		}
		custom_tooltip = utilitarian_timeout_fail_tt
		custom_tooltip = utilitarian_timeout_success_modifiers_tt
		custom_tooltip = utilitarian_timeout_success_ideology_tt
		set_variable = utilitarianism_done
	}

	on_yearly_pulse = {
		effect = {

		}
		random_events = {
			100 = 0
			10 = utilitarian.4
			10 = utilitarian.5
			10 = utilitarian.6
			10 = utilitarian.7
			# utilitarian.8 is fired by the enactment pulse
			10 = utilitarian.10
		}
	}

	timeout = 10950

	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = yes

}
