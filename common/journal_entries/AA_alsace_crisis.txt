je_crisis_alsace_france = {
	group = je_group_internal_affairs
	icon = "gfx/interface/icons/event_icons/fra.dds"

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		c:FRA = ROOT
    }	

	possible = 
	{	
		c:FRA ?= THIS
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			Year >= 1871
			NOT = { exists = c:GER }
			custom_tooltip = {
				text = ger_has_fail
				NOT = { exists = c:GER }
			}
			is_at_war = no
			owns_entire_state_region = STATE_ALSACE_LORRAINE
		}
		effect = {
			trigger_event = { id = dsk_france.1 }
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			c:GER = { owns_entire_state_region = STATE_ALSACE_LORRAINE }
			custom_tooltip = {
				text = ger_has_formed
				exists = c:GER
			}
		}
		effect = {
			trigger_event = { id = dsk_france.2 }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = ally_fra_pru
				ROOT =
				{
					OR =
					{
						any_scope_treaty = {
							binds = c:PRU
							any_scope_article = {
								has_type = alliance
							}
						}
						any_scope_treaty = {
							binds = c:NGF
							any_scope_article = {
								has_type = alliance
							}
						}
						any_scope_treaty = {
							binds = c:GER
							any_scope_article = {
								has_type = alliance
							}
						}
					}
				}
			}
		}
	}

	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
}
	
je_crisis_alsace_germany = {
	group = je_group_internal_affairs
	icon = "gfx/interface/icons/event_icons/ger.dds"

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		c:PRU ?= THIS

    }	
		
	possible = 
	{
		OR = 
		{
			c:PRU ?= THIS
			c:NGF ?= THIS
			c:GER ?= THIS		
		}
	
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			Year <= 1871
			owns_entire_state_region = STATE_ALSACE_LORRAINE
			custom_tooltip = {
				text = ger_alive
				c:GER ?= THIS
			}
		}
		effect = {
			trigger_event = { id = dsk_france.4	}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = { owns_entire_state_region = STATE_ALSACE_LORRAINE }
			OR =
			{
				custom_tooltip = {
					text = ger_alive
					c:GER ?= THIS
				}
				Year >= 1871
			}
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = ally_fra_pru
				ROOT =
				{
					any_scope_treaty = {
						binds = c:FRA
						any_scope_article = {
							has_type = alliance
						}
					}
				}
			}
		}
	}

	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
}
