# Unification

REPLACE_OR_CREATE:ai_strategy_unify_india = {
	icon = "gfx/interface/icons/ai_strategy_icons/unification.dds"

	type = diplomatic

	will_form_power_bloc = {
		always = yes
	}

	undesirable_infamy_level = {
		value = 50
	}

	unacceptable_infamy_level = {
		value = 100
	}

	ideological_opinion_effect_mult = 0.5

	wargoal_maneuvers_fraction = {
		value = 0.2
	}

	diplomatic_play_neutrality = {
		value = 0
	}

	diplomatic_play_boldness = {
		value = 50
	}

	obligation_value = {
		value = 0

		if = {
			limit = {
				scope:target_country = {
					is_in_geographic_region = geographic_region_india
					has_indian_primary_culture = yes
				}
			}
			add = 25
		}
	}

	recklessness = {
		value = 1.0
	}

	aggression = {
		value = 0.5

		if = {
			limit = {
				scope:target_country = {
					any_scope_state = {
						is_in_geographic_region = geographic_region_india
					}
				}
			}
			add = 10.0
		}

		# Less aggressive against countries we're just antagonistic against, rather than belligerent/domineering
		if = {
			limit = {
				has_attitude = {
					who = scope:target_country
					attitude = antagonistic
				}
			}
			multiply = 0.25
		}
	}

	building_group_weights = {
		bg_army = 1.5
	}

	state_value = {
		if = {
			limit = {
				scope:target_state = {
					is_in_geographic_region = geographic_region_india
				}
				has_strategic_land_adjacency = scope:target_state
			}
			add = 100
		}
	}

	wargoal_scores = {
		annex_country = {
			if = {
				limit = {
					scope:target_country = {
						is_in_geographic_region = geographic_region_india
					}
					has_strategic_land_adjacency = scope:target_country
				}
				add = 300
			}
		}
		conquer_state = {
			if = {
				limit = {
					scope:target_state = {
						is_in_geographic_region = geographic_region_india
					}
					has_strategic_land_adjacency = scope:target_state
				}
			}
			add = 250
		}
	}

	secret_goal_scores = {
		befriend = {
			trigger = {
				NOT = { scope:target_country = { is_subject_of = root } }
				scope:target_country = {
					is_in_geographic_region = geographic_region_india
				}
			}
			score = {
				value = 150

				if = {
					limit = {
						NOT = {
							has_strategic_land_adjacency = scope:target_country
						}
					}
					add = 150
				}
			}
		}
		conquer = {
			trigger = {
				NOT = { scope:target_country = { is_subject_of = root } }
				scope:target_country = {
					any_scope_state = {
						is_in_geographic_region = geographic_region_india
					}
				}
				has_strategic_land_adjacency = scope:target_country
			}
			score = {
				value = 500
			}
		}
	}

	possible = {
		can_form_nation = BHT
		has_indian_primary_culture = yes
		OR = {
			NOT = {
				exists = c:BIC
			}
			c:BIC ?= {
				has_variable = sepoy_mutiny_failed
			}
		}
		has_technology_researched = pan-nationalism
		is_subject = no
	}

	weight = {
		value = 25

		add = {
			value = army_size
			multiply = 10
		}

		if = {
			limit = { NOT = { is_country_type = unrecognized } }
			multiply = 10.0
		}
		else_if = {
			limit = { country_rank = rank_value:unrecognized_major_power }
			multiply = 5.0
		}
		else_if = {
			limit = { country_rank = rank_value:unrecognized_regional_power }
			multiply = 3.0
		}
	}
}
