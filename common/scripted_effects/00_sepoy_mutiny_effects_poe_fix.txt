REPLACE_OR_CREATE:update_indian_instability = {
	set_bar_progress = {
		value = {
			value = 600
			subtract = "scope:journal_entry.scripted_bar_progress(sepoy_mutiny_progress_bar_bengal)"
			subtract = "scope:journal_entry.scripted_bar_progress(sepoy_mutiny_progress_bar_bombay)"
			subtract = "scope:journal_entry.scripted_bar_progress(sepoy_mutiny_progress_bar_madras)"
			add = {
				value = var:num_months
				multiply = 1.5
			}
			max = 600
		}
		name = sepoy_mutiny_progress_bar_indian_instability
	}
}

# Create Sepoy Rebellion Countries; call within c:BIC
REPLACE:sepoy_mutiny_immediate_effects = {
	# We will re-incorporate these states if BIC crushes the rebellion
	every_scope_state = {
		limit = {
			is_incorporated = yes
		}
		set_variable = bic_incorporated_state
	}

	# Player Instigators get to grab up to 2 states
	ordered_country = {
		limit = {
			is_player = yes
			has_variable = sepoy_mutiny_instigator
			is_in_geographic_region = geographic_region_india
			has_indian_primary_culture = yes
		}
		order_by = {
			value = prestige
			if = {
				limit = {
					has_variable = sepoy_mutiny_revolter_var
				}
				add = 50000
			}
			if = {
				limit = {
					has_variable = sepoy_mutiny_instigator
				}
				add = 100000
			}
		}
		while = {
			limit = {
				exists = c:BIC
				num_states < 2
				any_scope_state = {
					any_neighbouring_state = {
						owner = c:BIC
					}
				}
			}
			save_temporary_scope_as = state_eater_scope
			# Variant of eat_collapsing_bic_state that doesn't eat BIC's capital
			c:BIC ?= {
				ordered_scope_state = {
					limit = {
						any_neighbouring_state = {
							owner = scope:state_eater_scope
						}
						is_capital = no
					}
					order_by = {
						value = gdp
						multiply = 0.0001
						# Try to eat split states first
						if = {
							limit = {
								state_region ?= {
									any_scope_state = {
										owner ?= scope:state_eater_scope
									}
								}
							}
							add = 100000
						}
						# Then prioritze states in India
						if = {
							limit = {
								is_in_geographic_region = geographic_region_india
							}
							add = 70000
						}
						# Then ones with homelands of your culture
						if = {
							limit = {
								is_homeland_of_country_cultures = scope:state_eater_scope
							}
							add = 50000
						}
					}
					position = 0
					set_state_owner = scope:state_eater_scope
				}
			}
		}
	}
	set_local_variable = {
		name = bengal_num_soldiers
		value = 0
	}
	set_local_variable = {
		name = bengal_num_states
		value = 0
	}
	set_local_variable = {
		name = bombay_num_soldiers
		value = 0
	}
	set_local_variable = {
		name = bombay_num_states
		value = 0
	}
	set_local_variable = {
		name = madras_num_soldiers
		value = 0
	}
	set_local_variable = {
		name = madras_num_states
		value = 0
	}
	every_scope_state = {
		if = {
			limit = {
				is_in_bengal_presidency = yes
			}
			random_scope_building = {
				limit = {
					is_building_type = building_barrack
				}
				change_local_variable = {
					name = bengal_num_soldiers
					add = level
				}
			}
			change_local_variable = {
				name = bengal_num_states
				add = 1
			}
		}
		else_if = {
			limit = {
				is_in_bombay_presidency = yes
			}
			random_scope_building = {
				limit = {
					is_building_type = building_barrack
				}
				change_local_variable = {
					name = bombay_num_soldiers
					add = level
				}
			}
			change_local_variable = {
				name = bombay_num_states
				add = 1
			}
		}
		else_if = {
			limit = {
				is_in_madras_presidency = yes
			}
			random_scope_building = {
				limit = {
					is_building_type = building_barrack
				}
				change_local_variable = {
					name = madras_num_soldiers
					add = level
				}
			}
			change_local_variable = {
				name = madras_num_states
				add = 1
			}
		}
	}
	set_local_variable = {
		name = desired_bengal_rebelling_soldiers
		value = {
			value = local_var:bengal_num_soldiers
			multiply = {
				value = uneasy_raj_stability_medium_upper_bound
				subtract = global_var:bengal_presidency_stability
				divide = uneasy_raj_stability_medium_upper_bound
			}
		}
	}
	set_local_variable = {
		name = desired_bengal_states
		value = {
			value = local_var:bengal_num_states
			multiply = {
				value = uneasy_raj_stability_medium_upper_bound
				subtract = global_var:bengal_presidency_stability

			}
			min = 1
		}
	}
	set_local_variable = {

				value = uneasy_raj_stability_medium_upper_bound

	}
	set_local_variable = {

		value = {

			}
			min = 1
		}
	}
	set_local_variable = {
		name = desired_madras_rebelling_soldiers
		value = {

			multiply = {
				value = uneasy_raj_stability_medium_upper_bound
				subtract = global_var:madras_presidency_stability

			}
		}
	}
	set_local_variable = {
		name = desired_madras_states
		value = {
			value = local_var:madras_num_states
			multiply = {
				value = uneasy_raj_stability_medium_upper_bound
				subtract = global_var:madras_presidency_stability
				divide = uneasy_raj_stability_medium_upper_bound
			}
			min = 1
		}
	}
	while = {
		limit = {
			global_var:bengal_presidency_stability < uneasy_raj_stability_high_lower_bound
			local_var:desired_bengal_rebelling_soldiers > 0
			local_var:desired_bengal_states > 0
			c:BIC ?= {
				any_scope_state = {
					is_capital = no
					is_in_bengal_presidency = yes
				}
			}
		}
		ordered_scope_state = {
			limit = {
				is_capital = no
				is_in_bengal_presidency = yes
			}
			position = 0
			order_by = {
				value = radical_fraction
				if = {
					limit = {
						state_has_building_type_levels = {
			 				target = bt:building_barrack
			 				value >= local_var:desired_bengal_rebelling_soldiers
						}
					}
					add = 400
				}
				else_if = {
					limit = {
						state_has_building_type_levels = {
			 				target = bt:building_barrack
			 				value >= 20
						}
					}
					add = 300
				}
				else_if = {
					limit = {
						state_has_building_type_levels = {
			 				target = bt:building_barrack
			 				value >= 10
						}
					}
					add = 200
				}
				else_if = {
					limit = {
						state_has_building_type_levels = {
			 				target = bt:building_barrack
			 				value >= 5
						}
					}
					add = 100
				}
				scope:sepoy_scope ?= {
					if = {
						limit = {
							has_strategic_land_adjacency = prev
						}
						add = 1000
					}
					else_if = {
						limit = {
							has_strategic_adjacency = prev
						}
						add = 500
					}
				}
			}
			random_scope_building = {
				limit = {
					is_building_type = building_barrack
				}
				change_local_variable = {
					name = desired_bengal_rebelling_soldiers
					subtract = level
				}
			}
			change_local_variable = {
				name = desired_bengal_states
				subtract = 1
			}
			add_sepoy_cache_modifiers = yes
			if = {
				limit = {
					NOT = {
						exists = scope:sepoy_scope
					}
				}
				# Country gets saved to scope:sepoy_scope
				create_sepoy_mutiny_country = yes
				scope:sepoy_scope ?= {
					set_variable = bengal_mutiny
				}
			}
			else = {
				set_state_owner = scope:sepoy_scope
			}
		}
	}

		limit = {
			exists = scope:sepoy_scope
		}

			je:je_sepoy_mutiny ?= {
				add_involved_country = scope:sepoy_scope

					name = sepoy_mutiny

							has_strategic_land_adjacency = prev

						add = 1000
					}
					else_if = {
						limit = {
							has_strategic_adjacency = prev
						}
						add = 500
					}
				}
			}
			random_scope_building = {
				limit = {
					is_building_type = building_barrack
				}
				change_local_variable = {
					name = desired_bombay_rebelling_soldiers
					subtract = level
				}
			}
			change_local_variable = {
				name = desired_bombay_states
				subtract = 1
			}
			add_sepoy_cache_modifiers = yes
			if = {
				limit = {
					NOT = {
						exists = scope:sepoy_scope
					}
				}
				# Country gets saved to scope:sepoy_scope
				create_sepoy_mutiny_country = yes
				scope:sepoy_scope ?= {
					set_variable = bombay_mutiny
				}
			}
			else = {
				set_state_owner = scope:sepoy_scope
			}
		}
	}
	if = {
		limit = {
			exists = scope:sepoy_scope
		}
		c:BIC ?= {
			je:je_sepoy_mutiny ?= {
				add_involved_country = scope:sepoy_scope
			}
		}
		if = {
			limit = {
				NOT = {
					exists = scope:sepoy_mutiny_dp
				}
			}
			scope:sepoy_scope ?= {
				create_diplomatic_play = {
					name = sepoy_mutiny
					type = dp_sepoy_mutiny_new
					target_country = c:BIC
				}
			}
			random_diplomatic_play = {
				limit = {
					is_diplomatic_play_type = dp_sepoy_mutiny_new
				}
				save_scope_as = sepoy_mutiny_dp
			}
		}
		else = {
			scope:sepoy_mutiny_dp ?= {
				add_war_goal = {
					holder = scope:sepoy_scope
					type = sepoy_mutiny
					target_country = c:BIC
					primary_demand = yes
				}
			}
		}
		scope:sepoy_mutiny_dp ?= {
			add_war_goal = {
				holder = c:BIC
				type = annex_country
				target_country = scope:sepoy_scope
				primary_demand = yes
			}
		}
	}
	clear_saved_scope = sepoy_scope
	while = {
		limit = {
			global_var:madras_presidency_stability < uneasy_raj_stability_high_lower_bound
			local_var:desired_madras_rebelling_soldiers > 0
			local_var:desired_madras_states > 0
			c:BIC ?= {
				any_scope_state = {
					is_capital = no
					is_in_madras_presidency = yes
				}
			}
		}
		ordered_scope_state = {
			limit = {
				is_capital = no
				is_in_madras_presidency = yes
			}
			position = 0
			order_by = {
				value = radical_fraction
				if = {
					limit = {
						state_has_building_type_levels = {
			 				target = bt:building_barrack
			 				value >= local_var:desired_madras_rebelling_soldiers
						}
					}
					add = 400
				}
				else_if = {
					limit = {
						state_has_building_type_levels = {
			 				target = bt:building_barrack
			 				value >= 20
						}
					}
					add = 300
				}
				else_if = {
					limit = {
						state_has_building_type_levels = {
			 				target = bt:building_barrack
			 				value >= 10
						}
					}
					add = 200
				}
				else_if = {
					limit = {
						state_has_building_type_levels = {
			 				target = bt:building_barrack
			 				value >= 5
						}
					}
					add = 100
				}
				scope:sepoy_scope ?= {
					if = {
						limit = {
							has_strategic_land_adjacency = prev
						}
						add = 1000
					}
					else_if = {
						limit = {
							has_strategic_adjacency = prev
						}
						add = 500
					}

			random_scope_building = {

					is_building_type = building_barrack
				}
				change_local_variable = {
					name = desired_madras_rebelling_soldiers
					subtract = level
				}
			}

				name = desired_madras_states
				subtract = 1
			}

			if = {
				limit = {

						exists = scope:sepoy_scope

				}

				create_sepoy_mutiny_country = yes
				scope:sepoy_scope ?= {
					set_variable = madras_mutiny
				}
			}
			else = {
				set_state_owner = scope:sepoy_scope
			}
		}
	}
	if = {
		limit = {
			exists = scope:sepoy_scope
		}
		c:BIC ?= {
			je:je_sepoy_mutiny ?= {
				add_involved_country = scope:sepoy_scope
			}
		}
		if = {
			limit = {

REPLACE_OR_CREATE:add_sepoy_cache_modifiers = {
	set_local_variable = {
		name = barracks_level
		value = 0
	}
	if = {
		limit = {
			any_scope_building = {
				is_building_type = building_barrack
			}
		}
		random_scope_building = {
			limit = {
				is_building_type = building_barrack
			}
			set_local_variable = {
				name = barracks_level
				value = {
					value = level
					multiply = 2
				}
			}
		}
	}
	if = {
		limit = {
			local_var:barracks_level > 0
		}
		if = {
			limit = {
				NOT = {
					has_modifier = uprising_arms_cache_small_arms
				}
			}
			add_modifier = {
				name = uprising_arms_cache_small_arms
				multiplier = local_var:barracks_level
				years = 5
				is_decaying = yes
			}
		}
		if = {
			limit = {
				NOT = {
					has_modifier = uprising_arms_cache_ammunition
				}
			}
			add_modifier = {
				name = uprising_arms_cache_ammunition
				multiplier = local_var:barracks_level
				years = 5
				is_decaying = yes
			}
		}
		if = {
			limit = {
				NOT = {
					has_modifier = uprising_arms_cache_artillery 
				}
			}
			add_modifier = {
				name = uprising_arms_cache_artillery
				multiplier = local_var:barracks_level
				years = 5
				is_decaying = yes
			}
		}
	}
}

		}

					is_in_government = yes
				}
				remove_ruling_interest_group = yes
			}
		}
	}
