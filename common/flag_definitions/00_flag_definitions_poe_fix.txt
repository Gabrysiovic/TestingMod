REPLACE:BIC = { # British East India Company
	flag_definition = { # this is the raj flag
		coa = BIC
		coa_with_overlord_canton = BIC_subject
		allow_overlord_canton = yes
		priority = 100
		trigger = {
			scope:actor ?= {
				NOT = {
					has_government_type = gov_chartered_company
				}
			}
		}
	}
	flag_definition = { # this is the company flag
		coa = BIC_republic
		coa_with_overlord_canton = BIC_republic
		allow_overlord_canton = yes
		overlord_canton_scale = { 0.5 0.5 }
		priority = 10
		trigger = {
			scope:actor ?= {
				has_government_type = gov_chartered_company
			}
		}
	}

	flag_definition = { # Sepoy Mutiny against Raj
		coa = BIC_sepoy
		coa_with_overlord_canton = BIC_sepoy
		allow_overlord_canton = no
		priority = 1000
		trigger = {
			scope:actor ?= {
				has_variable = sepoy_mutiny_country
			}
			c:BIC ?= {
				NOT = {
					has_government_type = gov_chartered_company
				}
			}
		}
	}
	flag_definition = { # this is the company flag
		coa = BIC_republic_sepoy
		coa_with_overlord_canton = BIC_republic_sepoy
		allow_overlord_canton = no
		priority = 1001
		trigger = {
			scope:actor ?= {
				has_variable = sepoy_mutiny_country
			}
			c:BIC ?= {
				has_government_type = gov_chartered_company
			}
		}
	}
}

REPLACE:BHT = {
	flag_definition = {
		coa = BHT
		subject_canton = BHT
		priority = 1
		trigger = {
			scope:actor ?= {
				OR = {
					NOT = {
						was_formed_from = BIC
					}
					is_revolutionary = yes
					has_variable = india_home_rule_failed
					has_variable = indian_colonial_formed_by_revolution
					has_variable = indian_war_of_independence
				}
			}
		}
	}

	flag_definition = {
		coa = BHT_dominion_of_india
		coa_with_overlord_canton = BHT_dominion_of_india
		allow_overlord_canton = yes
		overlord_canton_scale = { 0.33 0.33 }
		priority = 10
		trigger = {
			scope:actor ?= {
				OR = {
					# This seems weird but we need it so the Indian National Movement uses the Dominion Flag
					c:BIC ?= this
					was_formed_from = BIC
				}
				NOR = {
					is_revolutionary = yes
					is_subject = no
					has_variable = india_home_rule_failed
					has_variable = indian_colonial_formed_by_revolution
					has_variable = indian_war_of_independence
				}
			}
		}
	}

	flag_definition = {
		coa = HND_communist
		priority = 1500
		trigger = {
			coa_def_communist_flag_trigger = yes
		}
	}

	flag_definition = {
		coa = HND_fascist
		subject_canton = HND_fascist
		priority = 100
		trigger = {
			scope:actor ?= {
				country_has_state_religion = rel:hindu
			}
			coa_def_fascist_flag_trigger = yes
		}
	}
}
