promote_immigration = {
    groups = { general }
    requires_approval = no
    should_notify_third_parties = yes

    selectable = {
        is_in_geographic_region = geographic_region_americas
        any_country = {
            is_in_geographic_region = geographic_region_europe
            ROOT.total_population < this.quarter_total_population
        }
        ROOT.total_population < immigration_population_cap
    }

    potential = {
        NOT = { is_country_type = decentralized }
        scope:target_country = { NOT = { is_country_type = decentralized } }
        is_revolutionary = no
        is_secessionist = no
        is_subject = no
    }

    possible = {
        has_diplomatic_relevance = scope:target_country
        scope:target_country = {
            custom_tooltip = {
                text = is_in_europe_tt
                is_in_geographic_region = geographic_region_europe
            }
            any_primary_culture = {
                culture_can_have_mass_migration_to_country = ROOT
            }
            any_scope_state = {
                is_incorporated = yes
                custom_tooltip = {
                    text = is_in_europe_tt
                    is_in_geographic_region = geographic_region_europe
                }
                count >= 3
            }
            trigger_if = {
                limit = {
                    has_variable_list = promote_immigration_recipient
                }
                custom_tooltip = {
                    text = "has_not_been_target_recently_tt"
                    NOT = {
                        any_in_list = {
                            variable = promote_immigration_recipient
                            ROOT ?= this
                        }
                    }
                }
            }
        }
        NOT = {
            any_scope_state = {
                is_mass_migration_target = yes
            }
        }
        NOT = {
            has_modifier = promote_immigration_expenses
        }
        any_scope_state = {
            is_incorporated = yes
            state_is_eligible_as_mass_migration_target = yes
        }
        custom_tooltip = {
            text = "has_enough_population_to_migrate_tt"
            ROOT.total_population < scope:target_country.quarter_total_population
        }
    }

    accept_effect = {
        scope:target_country = {
            random_scope_state = {
                limit = {
                    is_incorporated = yes
                    is_in_geographic_region = geographic_region_europe
                    any_scope_culture = {
                        has_homeland = PREV
                        is_primary_culture_of = scope:target_country
                        culture_can_have_mass_migration_to_country = ROOT
                    }
                }
                save_scope_as = migration_state
                random_scope_culture = {
                    limit = {
                        has_homeland = PREV
                        is_primary_culture_of = scope:target_country
                        culture_can_have_mass_migration_to_country = ROOT
                    }
                    save_scope_as = migration_culture
                }
            }
            add_to_variable_list = {
                name = promote_immigration_recipient
                target = ROOT
                days = normal_modifier_time
            }
        }

        ROOT = {
            ordered_scope_state = {
                limit = {
                    is_incorporated = yes
                    state_is_eligible_as_mass_migration_target = yes
                }
                order_by = migration_pull
                position = 0
                check_range_bounds = no
                save_scope_as = migration_target_state
            }
        }

        scope:migration_target_state = {
            create_mass_migration = {
                origin = scope:target_country
                culture = scope:migration_culture
            }
        }

        if = {
            limit  = {
                ROOT.total_population < scope:target_country.one_tenth_total_population
            }
            add_modifier = {
                name = promote_immigration_expenses
                multiplier = {
                    value = money_amount_multiplier_small
                    min = 1000
                }
                days = very_short_modifier_time
            }
        }
        else = {
            add_modifier = {
                name = promote_immigration_expenses
                multiplier = {
                    value = money_amount_multiplier_medium
                    min = 1000
                }
                days = very_short_modifier_time
            }
        }


    }

    ai = {
        evaluation_chance = {
            value = 0
        }

        propose_score = {
            value = 10
        }

        will_propose = {
            always = no
        }
    }
}
