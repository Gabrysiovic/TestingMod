REPLACE:sepoy_mutiny_progress_bar_bengal = {
 	name = "sepoy_mutiny_progress_bar_bengal_name"
 	desc = "sepoy_mutiny_progress_bar_bengal_desc"

 	default_green = yes

 	monthly_progress = {
		c:BIC ?= {
			if = {
				limit = {
					has_global_variable = bengal_presidency_stability
				}
				add = {
					desc = "bengal_progress_bar_moderating_factor_tt"
					value = indian_uprising_baseline_drift_target
					subtract = global_var:bengal_presidency_stability
					multiply = 0.01
					min = -0.25
				}
			}
			if = {
				limit = {
					any_scope_state = {
						is_in_bengal_presidency = yes
					}
				}
				subtract = {
					desc = "bengal_progress_from_radicals_tt"
					je:je_uneasy_raj ?= {
						add = var:bengal_avg_radicals
						multiply = 2.5
					}
				}
				add = {
					desc = "bengal_progress_from_loyalists_tt"
					je:je_uneasy_raj ?= {
						add = var:bengal_avg_loyalists
						multiply = 2.5
					}
				}
				add = {
					desc = "bengal_progress_from_sol_tt"
					je:je_uneasy_raj ?= {
						add = var:bengal_avg_sol
						subtract = 10
						divide = 2
					}
				}
			}
			add = {
				desc = "bengal_progress_from_liberty_desire_tt"
				je:je_uneasy_raj ?= {
					add = var:bengal_avg_liberty_desire
					subtract = 50
					divide = -50 # +1 if all at 0, -1 if all at 100
				}
			}
			if = {
				limit = {
					any_interest_group = {
						is_indian_interest_group = yes
						is_in_government = yes
					}
				}
				add = {
					every_interest_group = {
						limit = {
							is_indian_interest_group = yes
							is_in_government = yes
						}
						add = ig_clout
					}
					desc = "presidencies_progress_from_indian_igs_in_government_tt"
				}
			}
			add = {
				desc = "presidencies_progress_from_zamindars_tt"
				if = {
					limit = {
						ig:ig_landowners ?= {
							ig_approval <= angry
						}
					}
					value = -2
				}
				else_if = {
					limit = {
						ig:ig_landowners ?= {
							ig_approval <= unhappy
						}
					}
					value = -1
				}
				else_if = {
					limit = {
						ig:ig_landowners ?= {
							ig_approval >= happy
						}
					}
					value = 0.5
				}
				else_if = {
					limit = {
						ig:ig_landowners ?= {
							ig_approval >= loyal
						}
					}
					value = 1
				}
			}
			if = {
				limit = {
					exists = c:BIC.var:bengal_progress_from_events_var_add_4
					NOT = {
						c:BIC.var:bengal_progress_from_events_var_add_4 = 0
					}
				}
				add = {
					desc = "sepoy_mutiny_events_10_result_tt"
					value = c:BIC.var:bengal_progress_from_events_var_add_4
				}
			}
			if = {
				limit = {
					exists = c:BIC.var:bengal_progress_from_events_var_add_5
					NOT = {
						c:BIC.var:bengal_progress_from_events_var_add_5 = 0
					}
				}
				add = {
					desc = "raj_4_result_tt"
					value = c:BIC.var:bengal_progress_from_events_var_add_5
				}
			}
			if = {
				limit = {
					exists = c:BIC.var:bengal_progress_from_events_var_add_6
					NOT = {
						c:BIC.var:bengal_progress_from_events_var_add_6 = 0
					}
				}
				add = {
					desc = "raj_5_result_tt"
					value = c:BIC.var:bengal_progress_from_events_var_add_6
				}
			}
			# Military Wages
			if = {
				limit = { military_wage_level = very_low }
				add = {
					desc = "presidencies_very_low_military_wages_tt"
					subtract = 1
				}
			}
			if = {
				limit = { military_wage_level = low }
				add = {
					desc = "presidencies_low_military_wages_tt"
					subtract = 0.5
				}
			}
			if = {
				limit = { military_wage_level = high }
				add = {
					desc = "presidencies_high_military_wages_tt"
					add = 0.5
				}
			}
			if = {
				limit = { military_wage_level = very_high }
				add = {
					desc = "presidencies_very_high_military_wages_tt"
					add = 1
				}
			}
			# Taxation
			if = {
				limit = { tax_level = very_low }
				add = {
					desc = "presidencies_very_low_taxation_tt"
					add = 1
				}
			}
			if = {
				limit = { tax_level = low }
				add = {
					desc = "presidencies_low_taxation_tt"
					add = 0.5
				}
			}
			if = {
				limit = { tax_level = high }
				add = {
					desc = "presidencies_high_taxation_tt"
					subtract = 0.5
				}
			}
			if = {
				limit = { tax_level = very_high }
				add = {
					desc = "presidencies_very_high_taxation_tt"
					subtract = 1
				}
			}
			# Utilitarian Reforms Enacted
			subtract = {
				desc = "presidencies_utilitarian_reforms_tt"
				value = 0.25
				multiply = {
					value = 0
					if = {
						limit = {
							has_law_or_variant = law_type:law_compulsory_primary_school
						}
						add = 1
					}
					if = {
						limit = {
							has_law_or_variant = law_type:law_public_schools
						}
						add = 1
					}
					if = {
						limit = {
							NOT = {
								has_law_or_variant = law_type:law_no_womens_rights
							}
						}
						add = 1
					}
					if = {
						limit = {
							has_law_or_variant = law_type:law_slavery_banned
						}
						add = 1
					}
					if = {
						limit = {
							OR = {
								has_law_or_variant = law_type:law_proportional_taxation
								has_law_or_variant = law_type:law_graduated_taxation
							}
						}
						add = 1
					}
					if = {
						limit = {
							OR = {
								has_law_or_variant = law_type:law_hindu_caste_discrimination_outlawed
								has_law_or_variant = law_type:law_affirmative_action
							}
						}
						add = 1
					}
				}
			}
			# Religious laws
			if = {
				limit = {
					NOT = {
						has_law_or_variant = law_type:law_total_separation
					}
				}
				add = {
					desc = "presidencies_religious_laws_tt"
					subtract = 2
				}
			}
			# Military laws
			if = {
				limit = {
					NOT = {
						has_law_or_variant = law_type:law_professional_army
					}
				}
				add = {
					desc = "presidencies_military_laws_tt"
					subtract = 1
				}
			}
			# Famine
			if = {
				limit = {
					has_journal_entry = je_indian_famines
				}
				add = {
					desc = "presidencies_je_indian_famines_tt"
					subtract = 1
				}
			}
			# Subjects with je_princely_states (Fomenting Rebellion)
			if = {
				limit = {
					je:je_uneasy_raj ?= {
						var:bengal_fomenting_rebellion_progress > 0
					}
				}
				subtract = {
					desc = "presidencies_fomenting_rebellion_subjects_tt"
					je:je_uneasy_raj ?= {
						add = var:bengal_fomenting_rebellion_progress
						divide = 100
					}
				}
			}
		}
 	}

	start_value = 165
	min_value = 0
	max_value = 200
}
